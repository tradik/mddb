Quick Start #

bash — docker 
# Run MDDB server
docker run -d \
  -p 11023:11023 \
  -p 11024:11024 \
  -v mddb-data:/app/data \
  tradik/mddb:latest

# Test the API
curl http://localhost:11023/health
bash — docker compose 
# Clone repository
git clone https://github.com/tradik/mddb.git
cd mddb

# Start the server, the web panel and the chat server
docker compose up -d

# Services available (bound to 127.0.0.1):
# - MDDB HTTP: http://localhost:11023
# - MDDB gRPC: localhost:11024
# - MCP (HTTP): http://localhost:9000/mcp
# - Web Panel: http://localhost:3000
# - Chat:      ws://localhost:11030/ws
bash — binary 
# Download binary
wget https://github.com/tradik/mddb/releases/download/v{{$.Vars.mddbVersion}}/mddbd-v{{$.Vars.mddbVersion}}-linux-amd64.tar.gz
tar -xzf mddbd-v{{$.Vars.mddbVersion}}-linux-amd64.tar.gz

# Run server
./mddbd-v{{$.Vars.mddbVersion}}-linux-amd64/mddbd

# Server starts on http://localhost:11023
bash — build from source 
# Clone repository
git clone https://github.com/tradik/mddb.git
cd mddb

# Build (server and CLI)
make build

# Run
./services/mddbd/mddbd
{{end}}
